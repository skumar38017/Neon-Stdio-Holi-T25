# Creating the commands for setting up the database, role, and migrations.

# 1️⃣ Connect to the PostgreSQL Database
# Command to log in to PostgreSQL (password: postgres)
psql -h localhost -p 5432 -U postgres -d postgres
# password: postgres 

# 2️⃣ Create Role and Database
# i. Create a new role (username: Neon-Studioz-Holi-T25) with the specified password
CREATE ROLE "Neon-Studioz-Holi-T25" WITH LOGIN PASSWORD 'Neon-Studioz-Holi-T25';

# ii. Create a new database with the role as the owner
CREATE DATABASE "Neon-Studioz-Holi-T25" OWNER "Neon-Studioz-Holi-T25";

# iii. Grant all privileges on the database to the role
GRANT ALL PRIVILEGES ON DATABASE "Neon-Studioz-Holi-T25" TO "Neon-Studioz-Holi-T25";

# iv. Alter the role to grant superuser permissions and additional capabilities
ALTER ROLE "Neon-Studioz-Holi-T25" WITH SUPERUSER CREATEDB CREATEROLE REPLICATION BYPASSRLS;

# 3️⃣ Run Alembic Migrations (If Migrations Do Not Exist)
# Initialize alembic for migrations
alembic init alembic

# Open `alembic.ini` file and set the `target_metadata` to `Base.metadata`
# In `alembic.ini` file:
# [alembic]
# script_location = alembic
# sqlalchemy.url = <your_database_url>

# 4️⃣ Generate Initial Alembic Migration (If No Migrations Exist)
# This will generate the first migration file based on the schema
alembic revision --autogenerate -m "Initial migration"

# 5️⃣ Apply the Newly Generated Migrations to the Database
# This will apply the migrations and update the database schema
alembic upgrade head

# 6️⃣ Verify Migration Status
# List the history of applied migrations
alembic history

# --------------------------------------------------------------
# -------------------- Downgrade Section -----------------------
# --------------------------------------------------------------

# 7️⃣ Rollback the Last Migration (If Needed)
# This will undo the most recent migration
alembic downgrade -1

# 8️⃣ Remove a Specific Migration File (If Needed)
# Navigate to the migrations folder and remove a specific migration file
cd alembic/versions
rm b599ce210611_fix_delete_orphan_issue.py

# 9️⃣ Update Alembic Version Number Manually (If Necessary)
# If you need to reset the migration version manually
psql -U Neon-Studioz-Holi-T25 -d Neon-Studioz-Holi-T25
UPDATE alembic_version SET version_num = '041520bb1f59';

# List all tables in the database
psql -U Neon-Studioz-Holi-T25 -d Neon-Studioz-Holi-T25 -c "\dt"
